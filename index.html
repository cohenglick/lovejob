-- ===== טבלאות ===== create table if not exists jobs ( id uuid primary key, employer_id uuid not null references auth.users(id) on delete cascade, employer_name text, contact_email text, title text not null, description text, requirements text, scope text, location text, created_at timestamptz default now() ); create table if not exists candidates ( id uuid primary key, full_name text not null, address text, birth_date date, job_type text, scope text, salary numeric, availability text, has_video boolean default false, video_url text, has_resume boolean default false, resume_url text, resume_file_name text, is_active boolean default true, created_at timestamptz default now() ); create table if not exists applications ( id uuid primary key, job_id uuid not null references jobs(id) on delete cascade, candidate_id uuid not null references candidates(id) on delete cascade, status text default 'ממתין', notes text default '', created_at timestamptz default now() ); -- view ציבורי בלי סטטוס והערות פנימיות create or replace view public_applications as select id, job_id, candidate_id from applications; grant select on public_applications to anon, authenticated; -- ===== הרשאות (RLS) ===== alter table jobs enable row level security; alter table candidates enable row level security; alter table applications enable row level security; create policy "jobs read" on jobs for select to anon, authenticated using (true); create policy "jobs insert own" on jobs for insert to authenticated with check (auth.uid() = employer_id); create policy "jobs delete own" on jobs for delete to authenticated using (auth.uid() = employer_id); create policy "cand read" on candidates for select to anon, authenticated using (true); create policy "cand insert" on candidates for insert to anon, authenticated with check (true); create policy "cand update" on candidates for update to anon, authenticated using (true) with check (true); create policy "cand delete employers" on candidates for delete to authenticated using (true); create policy "app insert" on applications for insert to anon, authenticated with check (true); create policy "app read own jobs" on applications for select to authenticated using (exists (select 1 from jobs j where j.id = applications.job_id and j.employer_id = auth.uid())); create policy "app update own jobs" on applications for update to authenticated using (exists (select 1 from jobs j where j.id = applications.job_id and j.employer_id = auth.uid())); -- ===== אחסון לקורות חיים ווידאו ===== insert into storage.buckets (id, name, public) values ('media','media',true) on conflict (id) do nothing;
