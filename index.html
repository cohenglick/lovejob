create table jobs ( id uuid primary key default gen_random_uuid(), owner_id uuid not null default auth.uid() references auth.users(id) on delete cascade, company text not null, cv_email text not null, title text not null, description text not null, requirements text not null, scope text not null, city text not null, created_at timestamptz default now() ); create table profiles ( id uuid primary key default gen_random_uuid(), full_name text not null, city text not null, birth_date date not null, desired_role text not null, scope text not null, salary int not null, availability text not null, cv_url text, bio text, created_at timestamptz default now() ); create table applications ( id uuid primary key default gen_random_uuid(), job_id uuid not null references jobs(id) on delete cascade, profile_id uuid not null references profiles(id) on delete cascade, created_at timestamptz default now(), unique (job_id, profile_id) ); alter table jobs enable row level security; alter table profiles enable row level security; alter table applications enable row level security; create policy "jobs read" on jobs for select using (true); create policy "jobs insert" on jobs for insert to authenticated with check (owner_id = auth.uid()); create policy "jobs update" on jobs for update to authenticated using (owner_id = auth.uid()); create policy "jobs delete" on jobs for delete to authenticated using (owner_id = auth.uid()); create policy "profiles read" on profiles for select using (true); create policy "profiles insert" on profiles for insert with check (true); create policy "profiles update" on profiles for update using (true); create policy "apply" on applications for insert with check (true); create policy "employer reads applicants" on applications for select to authenticated using (exists (select 1 from jobs where jobs.id = job_id and jobs.owner_id = auth.uid())); insert into storage.buckets (id, name, public) values ('cvs', 'cvs', true); create policy "cv upload" on storage.objects for insert with check (bucket_id = 'cvs'); create policy "cv read" on storage.objects for select using (bucket_id = 'cvs')
